git init - иницилизация репозитория
git status - статус репозитория
git add file - добавить файл в индекс
git commit -m "" - зафиксировать изменения
git log - просмотреть историю коммитов
git branch "" - создать новую ветку
git checkout - перейти на ветку ( старые версии )
git switch - перейти на ветку ( новые версии )
git branch - показать список веток
git merge "" - слить указанную ветку в текущую
git branch -d "" - удалить ветку 
git remore add origin - ссылка на удаленный репозиторий
git push - отправить изменения на удаленный репозторий
git pull - полчить изменения из удаленного репозиторя
git clone "" - копировать удаленный репозиторий
